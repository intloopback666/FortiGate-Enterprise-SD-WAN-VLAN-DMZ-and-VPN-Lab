----LAN-SW01 configuration----

enable
configure terminal

hostname LAN-SW01
no ip domain-lookup

vlan 100
 name USERS_VLAN100

vlan 200
 name USERS_VLAN200

vlan 300
 name USERS_VLAN300

interface ethernet0/0
 description Uplink_to_FortiGate_port3
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 100,200,300
 spanning-tree portfast trunk

interface ethernet0/1
 description PC_VLAN100
 switchport mode access
 switchport access vlan 100
 spanning-tree portfast
 spanning-tree bpduguard enable

interface ethernet0/2
 description PC_VLAN200
 switchport mode access
 switchport access vlan 200
 spanning-tree portfast
 spanning-tree bpduguard enable

interface ethernet0/3
 description PC_VLAN300
 switchport mode access
 switchport access vlan 300
 spanning-tree portfast
 spanning-tree bpduguard enable

interface vlan 100
 ip address 10.100.0.2 255.255.255.0
 no shutdown

ip default-gateway 10.100.0.1

interface range ethernet0/1 - 3
 switchport port-security
 switchport port-security maximum 1
 switchport port-security violation restrict
 switchport port-security mac-address sticky

end
write memory

----ISP-1 configuration----

configure terminal

hostname ISP-1

interface ethernet0/0
 description To_FortiGate_port1
 ip address 172.16.1.1 255.255.255.252
 no shutdown

interface ethernet0/1
 description To_EVE_NG_Cloud_Bridge
 ip address 192.168.1.110
 no shutdown

ip access-list standard NAT_INSIDE
 permit 172.16.1.0 0.0.0.3

interface ethernet0/0
 ip nat inside

interface ethernet0/1
 ip nat outside

ip nat inside source list NAT_INSIDE interface ethernet0/1 overload



end
write memory
